function backup(name) {
    const p = path.join(ROOT, name);
    if (fs.existsSync(p)) {
        fs.copyFileSync(p, p + '.bak');
        console.log('  ✅ Backup: ' + name + '.bak');
    }
}

    const posts = db.prepare(\`SELECT id, slug, title_en, title_ar, title_ms,
        excerpt_en, excerpt_ar, excerpt_ms, image_url, category,
        published, featured, created_at, updated_at
        FROM blog_posts WHERE published = 1
        ORDER BY featured DESC, sort_order ASC, created_at DESC\`).all();
    res.json(posts);
});

app.get('/api/blog/:slug', (req, res) => {
    const post = db.prepare('SELECT * FROM blog_posts WHERE slug = ? AND published = 1').get(req.params.slug);
    if (!post) return res.status(404).json({ error: 'Post not found' });
    // Related posts
    const related = db.prepare(\`SELECT id, slug, title_en, title_ar, title_ms,
        excerpt_en, excerpt_ar, excerpt_ms, image_url, category, created_at
        FROM blog_posts WHERE published = 1 AND category = ? AND slug != ?
        ORDER BY created_at DESC LIMIT 3\`).all(post.category, req.params.slug);
    res.json({ ...post, related });
});

app.get('/api/admin/blog', authMiddleware, (req, res) => {
    const posts = db.prepare('SELECT * FROM blog_posts ORDER BY sort_order ASC, created_at DESC').all();
    res.json(posts);
});

app.post('/api/admin/blog', authMiddleware, (req, res) => {
    const b = req.body;
    const r = db.prepare(\`INSERT INTO blog_posts
        (slug, title_ar, title_en, title_ms, excerpt_ar, excerpt_en, excerpt_ms,
         content_ar, content_en, content_ms, image_url, category, published, featured, sort_order)
        VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)\`).run(
        b.slug, b.title_ar||'', b.title_en||'', b.title_ms||'',
        b.excerpt_ar||'', b.excerpt_en||'', b.excerpt_ms||'',
        b.content_ar||'', b.content_en||'', b.content_ms||'',
        b.image_url||'', b.category||'news', b.published||0, b.featured||0, b.sort_order||0);
    res.json({ id: r.lastInsertRowid, message: 'Created' });
});

app.put('/api/admin/blog/:id', authMiddleware, (req, res) => {
    const b = req.body;
    db.prepare(\`UPDATE blog_posts SET slug=?, title_ar=?, title_en=?, title_ms=?,
        excerpt_ar=?, excerpt_en=?, excerpt_ms=?, content_ar=?, content_en=?, content_ms=?,
        image_url=?, category=?, published=?, featured=?, sort_order=?, updated_at=CURRENT_TIMESTAMP
        WHERE id=?\`).run(
        b.slug, b.title_ar||'', b.title_en||'', b.title_ms||'',
        b.excerpt_ar||'', b.excerpt_en||'', b.excerpt_ms||'',
        b.content_ar||'', b.content_en||'', b.content_ms||'',
        b.image_url||'', b.category||'news', b.published||0, b.featured||0, b.sort_order||0,
        req.params.id);
    res.json({ message: 'Updated' });
});

app.delete('/api/admin/blog/:id', authMiddleware, (req, res) => {
    db.prepare('DELETE FROM blog_posts WHERE id=?').run(req.params.id);
    res.json({ message: 'Deleted' });
});

app.patch('/api/admin/blog/:id/toggle', authMiddleware, (req, res) => {
    const p = db.prepare('SELECT published FROM blog_posts WHERE id=?').get(req.params.id);
    if (!p) return res.status(404).json({ error: 'Not found' });
    db.prepare('UPDATE blog_posts SET published=?, updated_at=CURRENT_TIMESTAMP WHERE id=?').run(p.published ? 0 : 1, req.params.id);
    res.json({ published: !p.published });
});

    db.prepare('UPDATE blog_posts SET featured=?, updated_at=CURRENT_TIMESTAMP WHERE id=?').run(p.featured ? 0 : 1, req.params.id);
    res.json({ featured: !p.featured });
});

function renderBlogFull() {
    const grid = document.getElementById('blogFullGrid');
    if (!grid) return;
    const search = (document.getElementById('blogSearchInput')?.value || '').toLowerCase();
    const l = typeof lang !== 'undefined' ? lang : 'en';
    let filtered = blogPosts.filter(p => {
        if (blogFilter !== 'all' && p.category !== blogFilter) return false;
        if (search) {
            const t = (p['title_' + l] || '').toLowerCase();
            const e = (p['excerpt_' + l] || '').toLowerCase();
            if (!t.includes(search) && !e.includes(search)) return false;
        }
        return true;
    });
    let html = '';
    const feat = filtered.find(p => p.featured);
    if (feat) html += renderBlogCard(feat, true);
    filtered.filter(p => !p.featured || p.id !== (feat && feat.id)).forEach(p => {
        html += renderBlogCard(p, false);
    });
    grid.innerHTML = html || '<p style="text-align:center;grid-column:1/-1;color:#999">No articles found.</p>';
}

function setBlogFilter(cat, btn) {
    blogFilter = cat;
    document.querySelectorAll('.blog-filter-btn').forEach(b => b.classList.remove('active'));
    if (btn) btn.classList.add('active');
    renderBlogFull();
}

function filterBlogPosts() { renderBlogFull(); }

        document.getElementById('blogDetailContent').innerHTML = content;
        const imgWrap = document.getElementById('blogDetailImgWrap');
        if (post.image_url) {
            document.getElementById('blogDetailImg').src = post.image_url;
            imgWrap.style.display = 'block';
        } else {
            imgWrap.style.display = 'none';
        }
        const backBtns = { en: '← Back to Blog', ar: '→ العودة للمدونة', ms: '← Kembali ke Blog' };
        document.getElementById('blogBackBtn').textContent = backBtns[l] || backBtns.en;
        // Related
        const relSection = document.getElementById('blogRelatedSection');
        const relGrid = document.getElementById('blogRelatedGrid');
        const relTitle = { en: 'Related Articles', ar: 'مقالات ذات صلة', ms: 'Artikel Berkaitan' };
        document.getElementById('blogRelatedTitle').textContent = relTitle[l] || relTitle.en;
        if (post.related && post.related.length > 0) {
            relGrid.innerHTML = post.related.map(p => renderBlogCard(p, false)).join('');

        } else {
            relSection.style.display = 'none';

    } catch (e) { console.log('Blog detail error:', e); }
